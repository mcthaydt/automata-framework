/**
 * Colour, last: a lit surface's albedo × its light, the haze over it, and the one snap to DUEL, with a Bayer dither so a
 * colour between two palette colours lands on both, in a pattern, not on one of them in a flat patch. Every colour on a
 * frame goes through `snap` once, so it is a DUEL colour whatever came before.
 */
import { duel } from '../../lib/figure';
import type { TimeOfDay, V3 } from '../world/types';
import { LOOK, rgb01 } from './look';

/** A 4×4 Bayer matrix (0–15). */
export const BAYER: readonly number[] = [0, 8, 2, 10, 12, 4, 14, 6, 3, 11, 1, 9, 15, 7, 13, 5];

/** Albedo (a 24-bit colour) × light × gain, per channel in sRGB 0–1, clamped to 1. */
export function litColour(albedo: number, light: V3, gain: number): V3 {
  return [(albedo >> 16) & 255, (albedo >> 8) & 255, albedo & 255].map((v, c) => Math.min(1, (v / 255) * light[c] * gain)) as V3;
}
/** The haze's share at a texel `behind` metres past the focus: 0 before focus + near, rising linearly to `amount` at
 * focus + far. */
export function hazeShare(behind: number, h = LOOK.haze): number {
  return Math.min(1, Math.max(0, (behind - h.near) / (h.far - h.near))) * h.amount;
}
/** The haze at `time`: the profile's, with the time's own colour and amount where it sets them. */
export const hazeOf = (time: TimeOfDay, h = LOOK.haze) => (time.haze ? { ...h, colour: time.haze.colour ?? h.colour, amount: time.haze.amount ?? h.amount } : h);
/** A colour hazed: mixed toward the veil, whose colour glows with the texel's light (haze.colour + glow × light), by
 * `share`. */
export function hazed(c: V3, light: V3, share: number, h = LOOK.haze): V3 {
  if (share <= 0) return c;
  const veil = rgb01(h.colour);
  return c.map((v, i) => v + (Math.min(1, veil[i] + h.glow * light[i]) - v) * share) as V3;
}
/** The one snap: the colour plus the Bayer offset at (x, y) × strength, to its nearest DUEL colour (FigureBuilder's
 * `duel`, which caches by 24-bit colour). */
export function snap(c: V3, x: number, y: number, strength = LOOK.dither.strength): number {
  const offset = (BAYER[(x & 3) + ((y & 3) << 2)] / 16 - 0.5) * strength, q = (v: number) => Math.min(255, Math.max(0, Math.round((v + offset) * 255)));
  return duel((q(c[0]) << 16) | (q(c[1]) << 8) | q(c[2]));
}
