/**
 * The before-and-after sheet of the turned-face fix (task 4d). Run from extensions/figures/scripts/ (it imports the
 * painter's source by relative path), in three runs:
 *
 *   MODE=before tsx scripts/turned-face-sheet.ts   (the painter without the fix: writes /tmp/tf/before.json)
 *   MODE=after  tsx scripts/turned-face-sheet.ts   (the painter with the fix: writes /tmp/tf/after.json)
 *   MODE=sheet OUT=<dir> tsx scripts/turned-face-sheet.ts   (pairs every cell that differs, writes the PNGs and a table)
 *
 * It draws the frames of the base, actors and sprites goldens as the goldens' own helpers do (the actors' by a copy of
 * tests/golden/actors.ts that keeps the pixels in place of their md5; the copy is made by the run's caller).
 */
import { mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { createCanvas } from '@napi-rs/canvas';
import { BASE } from '../base/base';
import { hero, member, standing, walking } from '../src/features/figure/frames';
import { ACTOR_WHO, actorFrames } from './_actors-pixels';

type Cell = { w: number; h: number; data: string }
const PITCH = 18 * Math.PI / 180
const mode = process.env.MODE ?? ''
const pack = (w: number, h: number, px: Uint8ClampedArray): Cell => ({ w, h, data: Buffer.from(px.buffer, px.byteOffset, px.byteLength).toString('base64') })

function collect(): Record<string, Cell> {
  const out: Record<string, Cell> = {}
  // The actors golden: key `base-m/wave/1@D2`, the density in the key.
  for (const who of ACTOR_WHO) {
    const frames = actorFrames(who) as unknown as Record<string, Uint8ClampedArray>
    for (const [key, px] of Object.entries(frames)) {
      const d = Number(key.split('@D')[1]), def = BASE.find((c) => c.def.id === who)!.def, [w, h] = member(def, d).cell
      out[`actors ${key}`] = pack(w, h, px)
    }
  }
  for (const d of [1, 2]) for (const c of BASE) {
    const cast = member(c.def, d), [w, h] = cast.cell
    // The base golden: the sheets' frames at the sheets' pitch.
    ;[0, 1, 2, 3, 4, 5, 6, 7].forEach((v, i) => (out[`base ${c.def.id}-turn-d${d}/${i}`] = pack(w, h, standing(cast, v, PITCH))))
    walking(cast, PITCH).forEach((px, i) => (out[`base ${c.def.id}-walk-d${d}/${i}`] = pack(w, h, px)))
    out[`base ${c.def.id}-hero-d${d}/0`] = pack(w, h, hero(cast, c.pose, Math.PI / 4, PITCH))
    // The sprites golden: the default pitch; its set is the hero, the 8 views and the walk.
    const set = [hero(cast, c.pose), ...[0, 1, 2, 3, 4, 5, 6, 7].map((v) => standing(cast, v)), ...walking(cast)]
    set.forEach((px, i) => (out[`sprites ${c.def.id}-D${d}/${i}`] = pack(w, h, px)))
  }
  return out
}

const unpack = (c: Cell) => new Uint8ClampedArray(Buffer.from(c.data, 'base64'))

if (mode === 'before' || mode === 'after') {
  mkdirSync('/tmp/tf', { recursive: true })
  writeFileSync(`/tmp/tf/${mode}.json`, JSON.stringify(collect()))
  console.log(mode, 'written')
} else if (mode === 'sheet') {
  const out = process.env.OUT ?? '/tmp/tf/out', ZOOM = 8, PAD = 5, LABEL = 22
  mkdirSync(out, { recursive: true })
  const before = JSON.parse(readFileSync('/tmp/tf/before.json', 'utf8')) as Record<string, Cell>
  const after = JSON.parse(readFileSync('/tmp/tf/after.json', 'utf8')) as Record<string, Cell>
  const rows: Record<string, { key: string; w: number; h: number; a: Uint8ClampedArray; b: Uint8ClampedArray; box: [number, number, number, number]; raw: number; seen: number }[]> = {}
  for (const key of Object.keys(after)) {
    const A = after[key]!, B = before[key]!, a = unpack(A), b = unpack(B)
    let raw = 0, seen = 0, x0 = A.w, y0 = A.h, x1 = -1, y1 = -1
    for (let k = 0; k < A.w * A.h; k++) {
      const o = k * 4
      const differs = a[o] !== b[o] || a[o + 1] !== b[o + 1] || a[o + 2] !== b[o + 2] || a[o + 3] !== b[o + 3]
      if (!differs) continue
      raw++
      const visible = a[o + 3] !== b[o + 3] || (a[o + 3] !== 0 && (a[o] !== b[o] || a[o + 1] !== b[o + 1] || a[o + 2] !== b[o + 2]))
      if (visible) { seen++; const x = k % A.w, y = (k / A.w) | 0; x0 = Math.min(x0, x); x1 = Math.max(x1, x); y0 = Math.min(y0, y); y1 = Math.max(y1, y) }
    }
    if (raw === 0) continue
    const file = key.split(' ')[0] + '-' + (key.split(' ')[1] ?? '').split('/')[0]
    ;(rows[file] ??= []).push({ key, w: A.w, h: A.h, a, b, box: [x0, y0, x1, y1], raw, seen })
  }
  const table: string[] = []
  for (const [file, list] of Object.entries(rows)) {
    // One picture a golden (and the actors' per figure): every moved cell cropped about its differing texels.
    const crops = list.map((r) => {
      const [x0, y0, x1, y1] = r.box
      const cx0 = Math.max(0, x0 - PAD), cy0 = Math.max(0, y0 - PAD), cx1 = Math.min(r.w - 1, x1 + PAD), cy1 = Math.min(r.h - 1, y1 + PAD)
      return { r, cx0, cy0, cw: cx1 - cx0 + 1, ch: cy1 - cy0 + 1 }
    })
    const panelW = Math.max(...crops.map((c) => c.cw)) * ZOOM, width = Math.max(panelW * 3 + 40, 760)
    const height = crops.reduce((n, c) => n + c.ch * ZOOM + LABEL + 12, 8)
    const canvas = createCanvas(width, height), ctx = canvas.getContext('2d')
    ctx.fillStyle = '#2b2b2b'; ctx.fillRect(0, 0, width, height)
    let y = 8
    for (const { r, cx0, cy0, cw, ch } of crops) {
      ctx.fillStyle = '#fff'; ctx.font = '14px sans-serif'
      ctx.fillText(`${r.key}   (before | after | changed texels in red)   ${r.seen} visible of ${r.raw} texels, x ${r.box[0]}..${r.box[2]}, y ${r.box[1]}..${r.box[3]} of ${r.w}x${r.h}`, 8, y + 14)
      y += LABEL
      for (let p = 0; p < 3; p++) {
        const x = 10 + p * (panelW + 10)
        for (let j = 0; j < ch; j++) for (let i = 0; i < cw; i++) {
          const o = ((cy0 + j) * r.w + cx0 + i) * 4, src = p === 0 ? r.b : r.a
          const moved = r.a[o] !== r.b[o] || r.a[o + 1] !== r.b[o + 1] || r.a[o + 2] !== r.b[o + 2] || r.a[o + 3] !== r.b[o + 3]
          const alpha = src[o + 3]!
          ctx.fillStyle = '#555'; ctx.fillRect(x + i * ZOOM, y + j * ZOOM, ZOOM, ZOOM)
          if (p === 2 && moved && (r.a[o + 3] !== 0 || r.b[o + 3] !== 0)) { ctx.fillStyle = '#ff2a2a'; ctx.fillRect(x + i * ZOOM, y + j * ZOOM, ZOOM, ZOOM); continue }
          if (alpha === 0) continue
          ctx.fillStyle = `rgb(${src[o]},${src[o + 1]},${src[o + 2]})`; ctx.fillRect(x + i * ZOOM, y + j * ZOOM, ZOOM, ZOOM)
        }
      }
      y += ch * ZOOM + 12
      table.push(`| ${file} | \`${r.key.split(' ').slice(1).join(' ')}\` | ${r.seen} | ${r.raw} | x ${r.box[0]}..${r.box[2]}, y ${r.box[1]}..${r.box[3]} of ${r.w}×${r.h} |`)
    }
    writeFileSync(join(out, `${file}.png`), canvas.toBuffer('image/png'))
    console.log(file, list.length, 'cells,', width, 'x', height)
  }
  writeFileSync(join(out, 'table.md'), `| picture | cell | visible texels | texels in all | box |\n|---|---|---|---|---|\n${table.join('\n')}\n`)
}
