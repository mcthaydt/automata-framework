import { writeFileSync } from 'node:fs'
import { test } from '@playwright/test'
import { begin, stateOf, waitFor } from './driver.ts'

// Temporary measurement probe: a Chromium trace over 6 s of play, summed by event name per process and thread.
test.setTimeout(10 * 60_000)
test('trace play', async ({ page, browser }) => {
  await page.setViewportSize({ width: 844, height: 390 })
  await begin(page)
  await page.evaluate(() => globalThis.__lastLight?.skip())
  await waitFor(page, { cutscene: null }, 600_000)
  await waitFor(page, { tickAtLeast: (await stateOf(page)).tick + 90 }, 300_000)
  await browser.startTracing(page, { categories: ['gpu', 'viz', 'cc', 'toplevel', 'blink', 'v8', 'disabled-by-default-gpu.service'] })
  await page.waitForTimeout(6000)
  const buf = await browser.stopTracing()
  writeFileSync(process.env.PROBE_OUT as string, buf)
})
