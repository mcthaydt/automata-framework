import sys, colorsys
import numpy as np
from PIL import Image
# Mean and percentiles of Rec.709 luma (sRGB values, 0..255), share under 16, hue of the brightest 5%.
# A crop: the HUD rows are excluded by masking the top 12% (objective, debug line, minimap) of the frame.
def stats(path):
    a = np.asarray(Image.open(path).convert('RGB')).astype(float)
    h = a.shape[0]
    a = a[int(h*0.15):]          # skip the HUD band (objective, pause, original's debug text and minimap)
    y = 0.2126*a[...,0] + 0.7152*a[...,1] + 0.0722*a[...,2]
    flat = y.ravel()
    p5, p50, p95 = np.percentile(flat, [5, 50, 95])
    dark = (flat < 16).mean()
    top = a.reshape(-1,3)[flat >= np.percentile(flat, 95)].mean(0)
    hh, ll, ss = colorsys.rgb_to_hls(*(top/255))
    # warm share: pixels where r > b + 20 among the brightest 5%
    tops = a.reshape(-1,3)[flat >= np.percentile(flat, 95)]
    warm = ((tops[:,0] - tops[:,2]) > 20).mean()
    # edge darkness: mean luma of the outer 10% frame ring vs the middle
    H, W = y.shape
    ring = np.ones_like(y, bool); ring[int(H*.1):int(H*.9), int(W*.1):int(W*.9)] = False
    return dict(mean=flat.mean(), p5=p5, p50=p50, p95=p95, dark=dark, hue=hh*360, warm=warm, edge=y[ring].mean()/max(1e-3,y[~ring].mean()))
if __name__ == '__main__':
    for p in sys.argv[1:]:
        s = stats(p)
        print(f"{p:60s} mean {s['mean']:5.1f} p5 {s['p5']:5.1f} p50 {s['p50']:5.1f} p95 {s['p95']:5.1f} dark<16 {s['dark']*100:4.1f}% hue95 {s['hue']:4.0f} warm95 {s['warm']*100:4.0f}% edge/mid {s['edge']:.2f}")
